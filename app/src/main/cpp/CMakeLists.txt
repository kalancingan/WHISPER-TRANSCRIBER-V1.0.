cmake_minimum_required(VERSION 3.22.1)
project(whispertranscriber)
include(FetchContent)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggerganov/whisper.cpp.git GIT_TAG v1.7.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(whisper)
add_library(whisper-jni SHARED whisper-jni.cpp)
find_library(log-lib log)
target_link_libraries(whisper-jni PRIVATE whisper ${log-lib} android)
target_compile_options(whisper-jni PRIVATE -O3)
